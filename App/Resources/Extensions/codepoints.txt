use strict;
use warnings;
binmode STDIN, ':encoding(UTF-8)';
binmode STDOUT, ':encoding(UTF-8)';
local $/;
my $text = <STDIN> // '';
print join(' ', map { sprintf('U+%04X', ord) } split(//, $text));
